01Perintä ja uudelleenkäyttö

Boss perii Employee-luokan ominaisuudet ja käyttää kantaluokan konstruktoria.

public class Employee
{
    public string Name { get; set; }
    public double Salary { get; set; }

    public Employee(string name, double salary)
    {
        Name = name;
        Salary = salary;
    }
}

public class Boss : Employee
{
    public double Bonus { get; set; }

    public Boss(string name, double salary, double bonus)
        : base(name, salary)
    {
        Bonus = bonus;
    }
}

03Rajapinta ja toteutukset

Rajapinta määrittelee yhteisen toiminnan, jonka eri luokat toteuttavat omalla tavallaan.

public interface ITransaction
{
    string ShowTransaction();
    decimal Money { get; }
}

public class PaidWithCash : ITransaction
{
    public decimal Money { get; }

    public PaidWithCash(decimal money)
    {
        Money = money;
    }

    public string ShowTransaction() =>
        $"Cash payment: {Money:0.00} €";
}
